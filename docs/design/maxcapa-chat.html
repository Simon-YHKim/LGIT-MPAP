SELECT machine_id, AVG(uph) AS avg_uph, COUNT(*) AS n
FROM   uph_input_runtime_daily_model
WHERE  model = 'R53A'
  AND  date = CURRENT_DATE - INTERVAL '1 day'
GROUP BY machine_id
ORDER BY avg_uph DESC;
